CÓDIGO FINAL WEBHOOK (stripe/route.ts)
Para invoice.payment_succeeded / invoice.paid:
typescript// 1. Buscar usuario CON case-insensitive
const user = await tx.user.findFirst({
  where: { 
    email: {
      equals: invoice.customer_email?.trim(),
      mode: 'insensitive'
    }
  },
  include: { instructorProfile: true }
});

// 2. Si encuentra usuario, buscar suscripción
if (user?.instructorProfile) {
  const sub = await tx.instructorSubscription.findFirst({
    where: { 
      instructorId: user.instructorProfile.id,
      status: { in: ['ACTIVE', 'PAST_DUE'] }
    },
    orderBy: { createdAt: 'desc' }
  });
  
  // 3. Si encuentra suscripción, activar
  if (sub) {
    await tx.instructorSubscription.update({
      where: { id: sub.id },
      data: { 
        status: 'ACTIVE',
        expiresAt: new Date(Date.now() + 30 * 24 * 60 * 60 * 1000),
        stripeSubscriptionId: invoice.subscription
      }
    });
    
    // 4. CREAR TRANSACCIÓN CONTABLE
    await tx.transaction.create({
      data: {
        type: 'INSTRUCTOR_SUBSCRIPTION',
        amount: invoice.amount_paid / 100,
        currency: 'MXN',
        status: 'COMPLETED',
        instructorId: user.instructorProfile.id,
        userId: user.id,
        stripePaymentIntentId: invoice.id,
        description: `Suscripción - ${user.email}`
      }
    });
  }
}

// 5. Respuesta con diagnósticos
return NextResponse.json({
  received: true,
  version: "v5_diag_direct",
  timestamp: new Date().toISOString(),
  diagnostics: {
    user_found: !!user,
    profile_found: !!user?.instructorProfile,
    subscription_found: !!sub,
    status_updated: sub ? 'ACTIVE' : 'SKIPPED'
  }
});
